from flask import Flask, request, jsonify, render_template app = Flask(__name__) @app.route("/") def home(): return render_template("index.html") @app.route("/simulate", methods=["POST"]) def simulate(): try: data = request.get_json() # Get inputs from index.html solar = float(data.get("solarIrradiance", 0)) area = float(data.get("collectorArea", 0)) efficiency = float(data.get("collectorEfficiency", 0)) ambient_temp = float(data.get("ambientTemperature", 30)) selected_fluid = data.get("workingFluid", "Auto") # Basic validation if solar <= 0 or area <= 0 or efficiency <= 0: return jsonify({ "error": "Please enter valid solar irradiance, collector area and efficiency." }), 400 # ------------------------------------------------- # 1. Solar heat available # ------------------------------------------------- solar_power = solar * area heat_absorption = solar_power * (efficiency / 100) # Convert W to kW heat_absorption_kw = heat_absorption / 1000 # ------------------------------------------------- # 2. Select working fluid # ------------------------------------------------- if selected_fluid == "Auto": if solar >= 800 and ambient_temp >= 30: fluid = "R245fa" elif solar >= 500: fluid = "R134a" elif solar >= 300: fluid = "R123" else: fluid = "Isobutane" else: fluid = selected_fluid # ------------------------------------------------- # 3. Estimated ORC efficiency # ------------------------------------------------- fluid_efficiency = { "R245fa": 0.14, "R134a": 0.12, "R123": 0.15, "Isobutane": 0.13 } orc_efficiency = fluid_efficiency.get(fluid, 0.12) # Adjust slightly according to solar temperature temperature_factor = max( 0.7, min(1.1, 1 + (ambient_temp - 25) * 0.005) ) system_efficiency = orc_efficiency * temperature_factor * 100 # ------------------------------------------------- # 4. Net power output # ------------------------------------------------- net_power = heat_absorption_kw * (system_efficiency / 100) # ------------------------------------------------- # 5. Approximate pressures # ------------------------------------------------- evaporator_pressure = { "R245fa": 15.0, "R134a": 12.0, "R123": 10.0, "Isobutane": 20.0 } condenser_pressure = { "R245fa": 2.0, "R134a": 3.0, "R123": 1.5, "Isobutane": 4.0 } evap_pressure = evaporator_pressure.get(fluid, 10.0) cond_pressure = condenser_pressure.get(fluid, 2.0) # ------------------------------------------------- # Return results to index.html # ------------------------------------------------- return jsonify({ "optimalWorkingFluid": fluid, "heatAbsorptionRate": round(heat_absorption_kw, 3), "netPowerOutput": round(net_power, 3), "systemEfficiency": round(system_efficiency, 2), "evaporatorPressure": round(evap_pressure, 2), "condenserPressure": round(cond_pressure, 2) }) except Exception as e: return jsonify({ "error": str(e) }), 500 if __name__ == "__main__": app.run( host="0.0.0.0", port=5000 )
